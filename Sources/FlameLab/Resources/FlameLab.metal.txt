#include <metal_stdlib>
using namespace metal;
struct Params { float dt; float time; float fuel; float mode; float2 mouse; float2 mousePrev; float2 wind; float2 nozzle; float2 sourceAir; };
constexpr sampler sampleLinear(coord::normalized, address::clamp_to_edge, filter::linear);
constexpr sampler sampleNearest(coord::normalized, address::clamp_to_edge, filter::nearest);
float hash2(float2 p) { return fract(sin(dot(p,float2(127.1,311.7)))*43758.5453); }
float noise2(float2 p) { float2 i=floor(p),f=fract(p); f=f*f*(3-2*f); return mix(mix(hash2(i),hash2(i+float2(1,0)),f.x),mix(hash2(i+float2(0,1)),hash2(i+1),f.x),f.y); }
float2 uvAt(uint2 g, uint2 n) { return (float2(g) + 0.5) / float2(n); }
float curl(texture2d<float, access::sample> v, float2 uv, float2 px) {
    return 0.5 * (v.sample(sampleLinear, uv + float2(px.x,0)).y - v.sample(sampleLinear, uv - float2(px.x,0)).y
                - v.sample(sampleLinear, uv + float2(0,px.y)).x + v.sample(sampleLinear, uv - float2(0,px.y)).x);
}
kernel void advance(texture2d<float, access::sample> oldV [[texture(0)]], texture2d<float, access::sample> oldS [[texture(1)]],
                    texture2d<float, access::write> newV [[texture(2)]], texture2d<float, access::write> newS [[texture(3)]],
                    constant Params &p [[buffer(0)]], uint2 g [[thread_position_in_grid]]) {
    uint2 n = uint2(newV.get_width(), newV.get_height()); if (any(g >= n)) return;
    float2 uv = uvAt(g,n), px = 1.0 / float2(n);
    float2 v0 = oldV.sample(sampleLinear,uv).xy;
    float2 back = clamp(uv - p.dt * v0, px * 0.5, 1.0 - px * 0.5);
    float2 v = oldV.sample(sampleLinear,back).xy * exp(-0.65*p.dt);
    // x: temperature, y: fuel, z: fast reaction, w: short-lived luminous soot.
    float4 s = oldS.sample(sampleLinear,back) * exp(float4(-0.38,-0.18,-8.0,-5.0)*p.dt);
    // Bounded cross-flow mixing gives the reacting gas a finite width without
    // stretching the finished image or adding a separate silhouette.
    float4 lateral = (oldS.sample(sampleLinear, back + float2(px.x,0)) + oldS.sample(sampleLinear, back - float2(px.x,0))) * 0.5;
    s = mix(s, lateral, min(0.08, p.dt * (2.5 + 2.0*p.fuel)));
    float2 d = (uv - p.nozzle) * float2(1.0, 1.8);
    float source = exp(-dot(d,d)/(0.00034 + p.fuel*0.00016));
    float variation = 0.95 + 0.10 * noise2(float2(p.time*4.0, 2.3));
    s.x += source * (4.0 + 2.0*p.fuel) * p.dt;
    s.y += source * (1.5 + 6.0*p.fuel) * p.dt;
    v.y -= source * (1.2 + 3.1*p.fuel) * variation * p.dt;
    // Positive temperature rises; screen UV y points down.
    v.y -= min(s.x, 3.0) * (0.23 + 0.35*p.fuel) * p.dt;
    float c = curl(oldV,uv,px);
    float2 grad = float2(abs(curl(oldV,uv+float2(px.x,0),px))-abs(curl(oldV,uv-float2(px.x,0),px)),
                         abs(curl(oldV,uv+float2(0,px.y),px))-abs(curl(oldV,uv-float2(0,px.y),px)));
    float2 normal = grad / max(length(grad),0.0001);
    v += float2(normal.y,-normal.x)*clamp(c*0.7,-0.08,0.08)*p.dt;
    float2 path = p.mouse-p.mousePrev, q = uv-p.mousePrev;
    float along = clamp(dot(q,path)/max(dot(path,path),1e-8),0.0,1.0);
    float2 delta = (q-along*path)*float2(1.0,1.5);
    float influence = exp(-dot(delta,delta)/0.012);
    v += clamp(p.wind,-1.2,1.2)*influence*2.0*p.dt;
    float2 wake = uv - p.nozzle;
    v += clamp(p.sourceAir,-1.0,1.0)*exp(-dot(wake,wake)/0.025)*p.dt;
    // Simplified, bounded reaction. This is a visual model, not chemical kinetics.
    float ignition = smoothstep(0.11,0.30,s.x);
    float consumed = min(s.y, s.y*ignition*2.6*p.dt);
    s.y -= consumed; s.x += consumed*1.2;
    float rate = consumed/max(p.dt,0.001);
    s.z = mix(s.z, rate, min(1.0, 15.0*p.dt));
    // Soot is a short-lived visual tracer generated by the reaction, not temperature.
    s.w += rate * (0.55 + 0.2*p.fuel) * p.dt;
    s.x *= exp(-(1.35 - 0.85*p.fuel)*p.dt); s.z = min(s.z,3.0);
    s = clamp(s,0.0,4.0);
    v = clamp(v,float2(-1.0),float2(1.0));
    if(g.x == 0 || g.x == n.x-1) v.x = 0;
    if(g.y == n.y-1) v.y = min(v.y,0.0);
    newV.write(float4(v,0,0),g); newS.write(s,g);
}
kernel void divergence(texture2d<float, access::sample> v [[texture(0)]], texture2d<float, access::write> div [[texture(1)]], uint2 g [[thread_position_in_grid]]) {
    uint2 n=uint2(div.get_width(),div.get_height()); if(any(g>=n)) return;
    float2 uv=uvAt(g,n), px=1.0/float2(n);
    float d=(v.sample(sampleNearest,uv+float2(px.x,0)).x-v.sample(sampleNearest,uv-float2(px.x,0)).x)*float(n.x)
           +(v.sample(sampleNearest,uv+float2(0,px.y)).y-v.sample(sampleNearest,uv-float2(0,px.y)).y)*float(n.y);
    div.write(float4(d*0.5,0,0,0),g);
}
kernel void pressure(texture2d<float, access::sample> previous [[texture(0)]], texture2d<float, access::sample> div [[texture(1)]],
                     texture2d<float, access::write> next [[texture(2)]], uint2 g [[thread_position_in_grid]]) {
    uint2 n=uint2(next.get_width(),next.get_height()); if(any(g>=n)) return;
    float2 uv=uvAt(g,n), px=1.0/float2(n); float wx=float(n.x*n.x), wy=float(n.y*n.y);
    float x=previous.sample(sampleNearest,uv+float2(px.x,0)).x+previous.sample(sampleNearest,uv-float2(px.x,0)).x;
    float y=previous.sample(sampleNearest,uv+float2(0,px.y)).x+previous.sample(sampleNearest,uv-float2(0,px.y)).x;
    float value=(wx*x+wy*y-div.sample(sampleNearest,uv).x)/(2.0*(wx+wy));
    if(g.y==0) value=0; // open top
    next.write(float4(value,0,0,0),g);
}
kernel void project(texture2d<float, access::sample> v [[texture(0)]], texture2d<float, access::sample> pressureField [[texture(1)]],
                    texture2d<float, access::write> out [[texture(2)]], uint2 g [[thread_position_in_grid]]) {
    uint2 n=uint2(out.get_width(),out.get_height()); if(any(g>=n)) return;
    float2 uv=uvAt(g,n), px=1.0/float2(n);
    float2 grad=float2((pressureField.sample(sampleNearest,uv+float2(px.x,0)).x-pressureField.sample(sampleNearest,uv-float2(px.x,0)).x)*float(n.x),
                       (pressureField.sample(sampleNearest,uv+float2(0,px.y)).x-pressureField.sample(sampleNearest,uv-float2(0,px.y)).x)*float(n.y))*0.5;
    float2 result=clamp(v.sample(sampleNearest,uv).xy-grad,-1.0,1.0);
    if(g.x==0 || g.x==n.x-1) result.x=0;
    if(g.y==n.y-1) result.y=min(result.y,0.0);
    out.write(float4(result,0,0),g);
}
struct Vertex { float4 position [[position]]; float2 uv; };
vertex Vertex vertexMain(uint id [[vertex_id]]) {
    float2 p[3]={float2(-1,-1),float2(3,-1),float2(-1,3)};
    Vertex v; v.position=float4(p[id],0,1); v.uv=float2((p[id].x+1)*0.5,(1-p[id].y)*0.5); return v;
}
fragment float4 fragmentMain(Vertex in [[stage_in]], texture2d<float> state [[texture(0)]], texture2d<float> velocity [[texture(1)]], texture2d<float> footage [[texture(2)]], constant Params &p [[buffer(0)]]) {
    float2 uv=in.uv; float4 s=state.sample(sampleLinear,uv); float2 v=velocity.sample(sampleLinear,uv).xy;
    if(p.mode>0.5 && p.mode<1.5) { float t=clamp(s.x/1.8,0.0,1.0); return float4(t,t*t,0.15*t,1); }
    if(p.mode>1.5 && p.mode<2.5) return float4(clamp(0.5+v.x*2.0,0.0,1.0),clamp(0.5+v.y*2.0,0.0,1.0),0.2,1);
    if (p.mode > 3.5) {
        float height = 0.20 + 0.28 * p.fuel;
        float width = height * 0.39;
        float h = (p.nozzle.y - uv.y) / height;
        if (h < 0.0 || h > 1.0) return float4(0.0);
        // Sample the simulated air: displacement grows away from the fixed root.
        float localWind = p.mode < 4.5 ? 0.0 : velocity.sample(sampleLinear, float2(p.nozzle.x, p.nozzle.y - h * height * 0.65)).x;
        float shift = clamp(localWind * 0.16, -0.10, 0.10) * h * h;
        float2 q = float2((uv.x - p.nozzle.x - shift) / width + 0.5, 1.0-h);
        if (q.x <= 0.0 || q.x >= 1.0) return float4(0.0);
        float frame = p.mode < 4.5 ? 0.0 : abs(fmod(p.time * 18.0, 30.0) - 15.0);
        float f0 = floor(frame), f1 = min(f0 + 1.0, 15.0);
        float2 safe = clamp(q, float2(0.005,0.003), float2(0.995,0.997));
        float2 tile0 = float2(fmod(f0,4.0),floor(f0/4.0));
        float2 tile1 = float2(fmod(f1,4.0),floor(f1/4.0));
        float3 rgb = mix(footage.sample(sampleLinear,(tile0+safe)/4.0).rgb,
                         footage.sample(sampleLinear,(tile1+safe)/4.0).rgb,fract(frame));
        // Remove the black photographic backdrop while retaining coloured light.
        float peak = max(rgb.r,max(rgb.g,rgb.b));
        float matte = smoothstep(0.025,0.20,peak);
        float edge = smoothstep(0.0,0.02,h) * (1.0-smoothstep(0.97,1.0,h));
        float alpha = clamp(peak*matte*edge,0.0,1.0);
        return float4(min(rgb*matte*edge,float3(alpha)),alpha);
    }
    float heat=s.x, reaction=s.z;
    if(p.mode>2.5) {
        float2 d=uv-p.nozzle; float h=-d.y;
        float n=noise2(float2(uv.x*12, uv.y*9+p.time*1.4));
        float w=(0.025+0.02*p.fuel)*smoothstep(0.0,0.05,h)*smoothstep(0.52,0.03,h);
        heat=max(0.0,(1.0-abs(d.x+(n-0.5)*0.025)/max(w,0.001)))*smoothstep(0.0,0.04,h)*smoothstep(0.55,0.10,h)*(1.0+p.fuel);
        float intensity = clamp(heat*0.75,0.0,1.0);
        float alpha = smoothstep(0.04,0.60,intensity)*0.82;
        float3 color = mix(float3(0.82,0.29,0.04), float3(1.0,0.84,0.45), smoothstep(0.05,0.85,intensity));
        return float4(color*alpha,alpha);
    }
    float fuel = max(0.0, s.y), soot = max(0.0, s.w);
    float2 px = 1.0 / float2(state.get_width(), state.get_height());
    float fuelEdge = length(float2(state.sample(sampleLinear,uv+float2(px.x,0)).y - state.sample(sampleLinear,uv-float2(px.x,0)).y,
                                  state.sample(sampleLinear,uv+float2(0,px.y)).y - state.sample(sampleLinear,uv-float2(0,px.y)).y));
    // Reaction + fresh fuel and local mixing select the blue combustion zone.
    // Its center remains dim where gas is concentrated but mixing is weak.
    float blue = smoothstep(0.012,0.20,reaction) * smoothstep(0.018,0.18,fuel)
               * (0.30 + 0.70*smoothstep(0.004,0.10,fuelEdge))
               * (1.0 - 0.90*smoothstep(0.015,0.065,soot));
    // Soot emission disappears quickly as reaction stops; heat alone cannot glow.
    float yellow = smoothstep(0.0005,0.035,soot) * smoothstep(0.045,0.25,heat)
                 * smoothstep(0.004,0.10,reaction);
    float flow = noise2(float2(uv.x*55.0,uv.y*34.0 + p.time*5.8));
    float detail = 0.91 + 0.18 * flow;
    yellow *= detail * (1.0 - 0.55 * blue);
    float blueAlpha = clamp(blue*0.78,0.0,0.78);
    float yellowAlpha = clamp(yellow*0.76,0.0,0.76);
    float alpha = 1.0 - (1.0-blueAlpha)*(1.0-yellowAlpha);
    float3 blueColor = mix(float3(0.11,0.18,0.82),float3(0.24,0.61,1.0),smoothstep(0.02,0.18,fuelEdge));
    float3 yellowColor = mix(float3(1.0,0.56,0.12),float3(1.0,0.91,0.62),smoothstep(0.12,0.48,heat));
    float3 premultiplied = blueColor*blueAlpha*(1.0-yellowAlpha*0.55) + yellowColor*yellowAlpha*(1.0-blueAlpha*0.55);
    return float4(min(premultiplied,float3(alpha)),alpha);
}
